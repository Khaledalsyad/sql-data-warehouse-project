INSERT INTO Silver.crm_cust_info(
cst_id,
cst_key,
cst_firstname,
cst_lastname,
cst_marital_status,
cst_gndr,
cst_create_date
)
select 
cst_id,
trim(cst_key),
trim(cst_firstname),
trim(cst_lastname),
CASE
	WHEN cst_marital_status = 'M'THEN UPPER(trim('maride'))
	WHEN cst_marital_status = 'S'THEN UPPER(trim('single'))
	ELSE 'a/n'
END as cst_marital_status,
CASE
	WHEN cst_gndr = 'M'THEN UPPER(trim('male'))
	WHEN cst_gndr = 'F'THEN UPPER(trim('female'))
	ELSE 'a/n'
END as cst_gndr,
cst_create_date
from (
select *,
ROW_NUMBER() over(partition by cst_id order by cst_create_date desc) as row_date
from bronze.crm_cust_info
where cst_id is not null
)x 
where row_date = 1

INSERT INTO  Silver.crm_prd_info(
prd_id,
prd_key,
prd_nm,
prd_cost,
prd_line,
prd_start_dt,
prd_end_dt
)

select 
prd_id,
REPLACE(left(prd_key, 5), '-', '_'),
SUBSTRING(prd_nm, 7, len(prd_nm)),
ISNULL(prd_cost, 0),
CASE 
WHEN prd_line = 'M' THEN UPPER(TRIM('MONATION'))
WHEN prd_line = 'R' THEN UPPER(TRIM('ROUDE'))
WHEN prd_line = 'S' THEN UPPER(TRIM('ATHER SALES'))
ELSE 'TOURING'
END AS prd_line,
cast(prd_start_dt as date),
LEAD(prd_start_dt) over(partition by prd_key order by prd_start_dt)  as prd_end_dt
from bronze.crm_prd_info

insert into Silver.crm_sales_details(
sls_ord_num,
sls_prd_key,
sls_cust_id,
sls_order_dt,
sls_ship_dt,
sls_due_dt,
sls_sales,
sls_quantity,
sls_price
)
select 
sls_ord_num,
sls_prd_key,
sls_cust_id,
case
when sls_order_dt = 0 or LEN(sls_order_dt) != 8 then null
else cast (cast(sls_order_dt as varchar) as date)
end as sls_order_dt,
case
when  sls_ship_dt= 0 or LEN(sls_ship_dt) != 8 then null
else cast (cast(sls_ship_dt as varchar) as date)
end as sls_ship_dt,
case
when sls_due_dt= 0 or LEN(sls_due_dt) != 8 then null
else cast (cast(sls_due_dt as varchar) as date)
end as sls_due_dt,
case 
when sls_sales is null or sls_sales <= 0 or sls_sales <> sls_quantity * sls_price
then sls_quantity * sls_price
else sls_sales
end,
case 
when sls_price <= 0 or sls_price is null 
then sls_sales / nullif(sls_price, 0)
else sls_price
end,
sls_quantity
from bronze.crm_sales_details

insert into Silver.erp_cust_az12(
cid,
bdate,
gen
)
select
case when cid like 'NASA%' then substring(cid, 4, len(cid))
else cid
end as cid,
bdate,
case 
when upper(trim(gen)) in ('F', 'Female')  then 'Female'
when upper(trim(gen)) in ('M', 'Male')  then 'Male'
else 'a/n'
end as gen
from bronze.erp_cust_az12
where bdate < '1924-01-01' or bdate  > GETDATE()
